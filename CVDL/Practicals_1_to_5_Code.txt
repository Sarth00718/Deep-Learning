def rotate_90(image):
    h, w = image.shape[:2]
    rotated = np.zeros_like(image)
    for y in range(h):
        for x in range(w):
            rotated[x, h-1-y] = image[y, x]
    return rotated

def rotate_180(image):
    h, w = image.shape[:2]
    rotated = np.zeros_like(image)
    for y in range(h):
        for x in range(w):
            rotated[h-1-y, w-1-x] = image[y, x]
    return rotated

def rotate_270(image):
    h, w = image.shape[:2]
    rotated = np.zeros_like(image)
    for y in range(h):
        for x in range(w):
            rotated[w-1-x, y] = image[y, x]
    return rotated

hist = [0] * 256

        hist[pixel_val] += 1

--- Code Block 3 ---


cdf = [0] * 256
cdf[0] = hist[0]
for i in range(1, 256):
    cdf[i] = cdf[i - 1] + hist[i]

cdf_min = min(val for val in cdf if val > 0)
total_pixels = rows * cols

normalized_cdf = [0] * 256
for i in range(256):
    normalized_cdf[i] = round((cdf[i] - cdf_min) / (total_pixels - cdf_min) * 255)

import cv2
import numpy as np
import matplotlib.pyplot as plt

img = cv2.imread('input.tif', cv2.IMREAD_GRAYSCALE)

bit_planes = []

plt.figure(figsize=(12, 9))

plt.subplot(3, 4, 1)
plt.imshow(img, cmap='gray')
plt.title('Original')
plt.axis('off')

for i in range(8):
    bit_mask = 1 << i
    bit_plane = cv2.bitwise_and(img, bit_mask)
    bit_plane_vis = np.where(bit_plane > 0, 255, 0).astype(np.uint8)
    bit_planes.append(bit_plane_vis)

    plt.subplot(3, 4, i + 2)
    plt.imshow(bit_plane_vis, cmap='gray')
    plt.title(f'Bit Plane {i}')
    plt.axis('off')

reconstructed = np.zeros_like(img)
for i in range(8):
    reconstructed += (bit_planes[i] // 255) * (1 << i)

plt.subplot(3, 4, 10)
plt.imshow(reconstructed, cmap='gray')
plt.title('ReDesigned')
plt.axis('off')

plt.tight_layout()
plt.show()
